function inputField(name, label) {

  return `
    <label>

      ${label}

      <input
        name="${name}"
        value="${String(ticket[name] ?? "")
          .replaceAll('"', "&quot;")}"
      >

    </label>
  `;
}

function saveTicket(event) {

  event.preventDefault();

  const fields =
    document.querySelectorAll(
      ".form input, .form select"
    );

  fields.forEach((field) => {

    if (field.name === "quantity") {
      ticket[field.name] = Number(field.value);
    } else {
      ticket[field.name] = field.value;
    }

  });

  save();

  showTickets();
}

/* =========================
   DEMO BARCODE
========================= */

function createBarcode() {

  let bars = "";

  for (let i = 0; i < 38; i++) {

    const height =
      75 + ((i % 5) * 5);

    bars += `
      <i
        class="bar"
        style="height:${height}%"
      ></i>
    `;
  }

  return bars;
}

function shareTicket() {

  const text =
    "I'm going to " +
    ticket.event +
    " — " +
    ticket.date +
    " at " +
    ticket.venue +
    ". DEMO ONLY.";

  if (navigator.share) {

    navigator.share({
      title: "Ticket Demo",
      text: text
    });

  } else {

    alert(text);

  }
}

function transferTicket() {

  A.innerHTML = `
    <div class="screen form">

      <h2>
        TRANSFER TICKETS
      </h2>

      <p>
        ${ticket.event}
      </p>

      <p>
        Section ${ticket.section}
        • Row ${ticket.row}
        • Seat ${ticket.seat}
      </p>

      <input
        id="firstName"
        placeholder="First Name *"
      >

      <input
        id="lastName"
        placeholder="Last Name *"
      >

      <input
        id="email"
        type="email"
        placeholder="Email Address *"
      >

      <textarea
        id="note"
        placeholder="Add an optional note"
      ></textarea>

      <button
        id="sendButton"
        class="primary"
        disabled
        onclick="createDemoTransfer()"
      >
        Forward ${ticket.quantity} Ticket
      </button>

      <p class="muted">

        This is a demo transfer.
        No real ticket or email is sent.

      </p>

    </div>
  `;

  const first =
    document.getElementById("firstName");

  const last =
    document.getElementById("lastName");

  const email =
    document.getElementById("email");

  const send =
    document.getElementById("sendButton");

  function checkForm() {

    send.disabled =
      !first.value.trim() ||
      !last.value.trim() ||
      !email.value.trim();

  }

  first.addEventListener(
    "input",
    checkForm
  );

  last.addEventListener(
    "input",
    checkForm
  );

  email.addEventListener(
    "input",
    checkForm
  );
}

function createDemoTransfer() {

  alert(
    "Demo transfer created. " +
    "No real email or ticket was sent."
  );

  showTickets();
}

/* =========================
   COUNTRY SELECTOR
========================= */

function country() {

  M.style.display = "flex";

  M.innerHTML = `

    <div class="sheet">

      <h2>
        Change Location
      </h2>

      ${countryList()}

      <br>

      <button
        class="secondary"
        onclick="closeCountry()"
      >
        Close
      </button>

    </div>

  `;
}

function countryList() {

  const countries = [

    ["🇺🇸", "United States"],
    ["🇬🇧", "United Kingdom"],
    ["🇪🇸", "España"],
    ["🇫🇷", "France"],
    ["🇩🇪", "Deutschland"],
    ["🇫🇮", "Suomi"],
    ["🇸🇪", "Sverige"],
    ["🇦🇪", "United Arab Emirates"]

  ];

  return countries
    .map((country) => {

      return `
        <button
          class="country"
          onclick="
            selectCountry(
              '${country[0]}'
            )
          "
        >
          ${country[0]}
          ${country[1]}
        </button>
      `;

    })
    .join("");
}

function selectCountry(flag) {

  document.getElementById(
    "flag"
  ).textContent = flag;

  closeCountry();
}

function closeCountry() {

  M.style.display = "none";
}
